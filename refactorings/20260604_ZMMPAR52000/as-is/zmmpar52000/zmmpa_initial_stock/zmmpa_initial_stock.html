FUNCTION ZMMPA_INITIAL_STOCK .
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(IV_BUDAT) TYPE BUDAT
*" TABLES
*" IT_ITEM STRUCTURE ZMMPAS51006 OPTIONAL
*" ET_ITEM STRUCTURE ZMMPAS51006 OPTIONAL
*"----------------------------------------------------------------------
*----------------------------------------------------------------------
* 기초 재고 구하는 함수
*----------------------------------------------------------------------
* Global data declarations

  DATA: LV_STOCK_QTY   TYPE NSDM_STOCK_QTY,
        LV_STOCK_DMBTR TYPE NSDM_DMBTR_STOCK,  " C01(+)
        LV_BUDAT_FR    TYPE BUDAT,
        LV_BUDAT_TO    TYPE BUDAT.
  TYPES: BEGIN OF TY_MAT,
           WERKS       TYPE MATDOC-WERKS,
           MATNR       TYPE MATDOC-MATNR,
           LGORT       TYPE MATDOC-LGORT,
           CHARG       TYPE MATDOC-CHARG,
           LIFNR       TYPE MATDOC-LIFNR,
           BWART       TYPE MATDOC-BWART,
           STOCK_QTY   TYPE MATDOC-STOCK_QTY,
           DMBTR_STOCK TYPE MATDOC-DMBTR_STOCK,
           SALK3       TYPE MATDOC-SALK3,
           LBKUM       TYPE MATDOC-LBKUM,
         END OF TY_MAT.


  CLEAR : ET_ITEM, ET_ITEM[].
  LV_BUDAT_FR = '19900101'.
  LV_BUDAT_TO = IV_BUDAT - 1.  " 프로그램에서 입력한 기준일자


  DATA(LT_DATA) = IT_ITEM[].


* SELECT A~WERKS,
* A~MATNR,
* A~LGORT,
* A~CHARG,
* SUM( STOCK_QTY ) AS EV_STOCK_QTY,
* SUM( DMBTR_STOCK ) AS EV_STOCK_DMBTR
* FROM MATDOC AS A JOIN ZMMPAT52010 AS B
* ON B~BWART EQ A~BWART
* JOIN @LT_DATA AS C
* ON A~WERKS EQ C~WERKS
* AND A~MATNR EQ C~MATNR
* AND A~LGORT EQ C~LGORT
* AND A~CHARG EQ C~CHARG
* AND A~LGORT NE ''
* AND C~LIFNR EQ ''
* WHERE A~BUDAT BETWEEN @LV_BUDAT_FR AND @LV_BUDAT_TO
* GROUP BY A~WERKS, A~MATNR, A~LGORT, A~CHARG
* INTO CORRESPONDING FIELDS OF TABLE @ET_ITEM.


  SELECT A~MJAHR,
         A~MBLNR,
         A~ZEILE,
         A~WERKS,
         A~MATNR,
         A~LGORT,
         A~CHARG,
         A~BWART,
         A~STOCK_QTY,
         A~DMBTR_STOCK
    FROM MATDOC AS A JOIN ZMMPAT52010 AS B
                       ON B~BWART EQ A~BWART
                     JOIN @LT_DATA AS C
                       ON A~WERKS EQ C~WERKS
                      AND A~MATNR EQ C~MATNR
                      AND A~LGORT EQ C~LGORT
                      AND A~CHARG EQ C~CHARG
                      AND A~LGORT NE ''
                      AND C~LIFNR EQ ''
   WHERE A~BUDAT BETWEEN @LV_BUDAT_FR AND @LV_BUDAT_TO
    INTO TABLE @DATA(LT_SUM1).

  SELECT A~MJAHR,
         A~MBLNR,
         A~ZEILE,
         A~WERKS,
         A~MATNR,
         A~LGORT_CID AS LGORT,
         A~CHARG_CID AS CHARG,
         0 AS STOCK_QTY,
         A~DMBTR AS DMBTR_STOCK
  FROM MATDOC AS A JOIN ZMMPAT52010 AS B
                     ON B~BWART EQ A~BWART
                   JOIN @LT_SUM1 AS C
                     ON A~MJAHR       = C~MJAHR
                    AND A~MBLNR       = C~MBLNR
                    AND A~ZEILE       = C~ZEILE
                    AND A~MATNR       = C~MATNR
                    AND A~WERKS       = C~WERKS
                    AND A~LGORT_CID   = C~LGORT
                    AND A~CHARG_CID   = C~CHARG
                    and c~bwart       = '101'
                    and c~DMBTR_STOCK = 0
 WHERE A~RECORD_TYPE = 'MDOC_CP'
   AND A~LBBSA_SID   = '06'
   AND A~DMBTR NE 0
  INTO TABLE @DATA(LT_SUM2).
  SORT LT_SUM2 BY MJAHR MBLNR ZEILE MATNR WERKS LGORT CHARG.


  LOOP AT LT_SUM1 INTO DATA(LS_SUM1).
    CLEAR ET_ITEM.
    IF LS_SUM1-BWART = '101' AND LS_SUM1-DMBTR_STOCK = 0.
      READ TABLE LT_SUM2 INTO DATA(LS_SUM2) WITH KEY MJAHR = LS_SUM1-MJAHR
                                                     MBLNR = LS_SUM1-MBLNR
                                                     ZEILE = LS_SUM1-ZEILE
                                                     MATNR = LS_SUM1-MATNR
                                                     WERKS = LS_SUM1-WERKS
                                                     LGORT = LS_SUM1-LGORT
                                                     CHARG = LS_SUM1-CHARG
                                                     BINARY SEARCH.
      IF SY-SUBRC = 0.
        LS_SUM1-DMBTR_STOCK = LS_SUM2-DMBTR_STOCK.
      ENDIF.
    ENDIF.


    MOVE-CORRESPONDING LS_SUM1 TO ET_ITEM.
    ET_ITEM-EV_STOCK_QTY   = LS_SUM1-STOCK_QTY.
    ET_ITEM-EV_STOCK_DMBTR = LS_SUM1-DMBTR_STOCK.

    COLLECT ET_ITEM.
    CLEAR LS_SUM2.
  ENDLOOP.

         A~CHARG,
         A~LIFNR_SID AS LIFNR,
         SUM( STOCK_QTY ) AS EV_STOCK_QTY,
         SUM( DMBTR_STOCK ) AS EV_STOCK_DMBTR
    FROM MATDOC AS A JOIN ZMMPAT52010 AS B
                       ON B~BWART EQ A~BWART
                     JOIN @LT_DATA AS C
                       ON A~WERKS     EQ C~WERKS
                      AND A~MATNR     EQ C~MATNR
                      AND A~LIFNR_SID EQ C~LIFNR
                      AND A~CHARG     EQ C~CHARG
                      AND A~LGORT     EQ ''
                      AND C~LIFNR     NE ''
   WHERE A~BUDAT BETWEEN @LV_BUDAT_FR AND @LV_BUDAT_TO
* AND A~CANCELLED = ''
* AND A~CANCELLATION_TYPE = ''
     AND A~SOBKZ = 'O'
   GROUP BY A~WERKS, A~MATNR, A~CHARG, A~LIFNR_SID
   APPENDING CORRESPONDING FIELDS OF TABLE @ET_ITEM.

ENDFUNCTION.
